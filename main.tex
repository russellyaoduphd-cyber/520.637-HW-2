\section{}
\subsection*{a} 
There are $N$ stages, where stage $k$ corresponds to parking space
$k$, for $k\in\{0,1,\ldots,N-1\}$.
Let $F$ denote that the current parking space is free, $F^c$ denote
that it is taken, and $T$ denote the terminal state in which the driver
has already parked. Then
\[\mathcal S_0=\{F,F^c\},\]
\[\mathcal S_k=\{F,F^c,T\}, \qquad k=1,\ldots,N-1, \]
and after the last decision stage,
\[\mathcal S_N=\{T\}. \]
For $k=0,\ldots,N-1$, the action sets are
\[ \mathcal A_k(F)=\{\text{park},\text{continue}\}, \]
\[\mathcal A_k(F^c)=\{\text{continue}\}. \]
At the terminal state $T$, no further decision is required.
\subsection*{b}
Let $a^1$ denote ``park'' and $a^2$ denote ``continue''. Let $C_{k+1}$ denote the incurred cost.
we have:\[P(S_{k+1}=T, C_{k+1}=c(k)|S_k=F,A_k=a^1)=1 \quad k= 0,1,\dots,{N-1}\]
Then we consider the state-action pair with $a^2$, for:
\[k=0,1,\ldots,{N-2}\]
\[P(S_{k+1}=F, C_{k+1}=0|S_k=F, A_k=a^2)=p(k+1)\]
\[P(S_{k+1}=F^c, C_{k+1}=0|S_k=F, A_k=a^2)=1-p(k+1)\]
\[P(S_{k+1}=F, C_{k+1}=0|S_k=F^c, A_k=a^2)=p(k+1)\]
\[P(S_{k+1}=F^c, C_{k+1}=0|S_k=F^c, A_k=a^2)=1-p(k+1)\]
For $k=N-1$, if the driver continues, then
\[P(S_N=T, C_N=C|S_{N-1}=F,A_{N-1}=a^2)=1\]
\[P(S_N=T, C_N=C|S_{N-1}=F^c,A_{N-1}=a^2)=1\]
All other transition-cost probabilities are zero.
\subsection*{c}
\[J^{\star}_N(T)=0\]
for $k={N-1}$, \[J^{\star}_{N-1}(F)=\min \{c(N-1),C\}\]
\[J^{\star}_{N-1}(F^c)=C\]
\[J^{\star}_{N-1}(T)=0\]
for $k=0,1,\ldots, {N-2}$,
\[J^{\star}_{k}(F)=\min\{c(k),p(k+1) J^{\star}_{k+1}(F)+(1-p(k+1))J^{\star}_{k+1}(F^c) \}\]
\[J^{\star}_{k}(F^c)= p(k+1) J^{\star}_{k+1}(F)+(1-p(k+1))J^{\star}_{k+1}(F^c)\]
\[J^{\star}_{k}(T)=0 \quad k= 1,2,\ldots,{N-2}\]
These equations are evaluated backward for
$k=N-1,N-2,\ldots,0$.
\[J^\star = p(0)J_0^\star(F) + (1-p(0))J_0^\star(F^c). \]
\subsection*{d}
\begin{proof}
Define
\[
V^{\star}_k = p(k) J^{\star}_{k}(F)+(1-p(k))J^{\star}_{k}(F^c),
\qquad k=0,1,\ldots,N-1,
\]
and, for convenience, let
\[
V_N^{\star}=C.
\]
From part (c), for $k=0,1,\ldots,N-2$,
\[
J_k^{\star}(F^c)=V_{k+1}^{\star},
\qquad
J_k^{\star}(F)=\min\{c(k),V_{k+1}^{\star}\}.
\]
For $k=N-1$, the same formulas hold because
\[
J_{N-1}^{\star}(F^c)=C=V_N^{\star},
\qquad
J_{N-1}^{\star}(F)=\min\{c(N-1),C\}.
\]
Hence, for every $k=0,1,\ldots,N-1$,
\[
\begin{aligned}
V_k^{\star}
&=p(k)\min\{c(k),V_{k+1}^{\star}\}
 +(1-p(k))V_{k+1}^{\star} \\
&\leq V_{k+1}^{\star}.
\end{aligned}
\]
Thus,
\[
V_0^{\star}\leq V_1^{\star}\leq\cdots\leq V_N^{\star}.
\]

If space $k$ is free, the driver should park whenever
\[
c(k)\leq V_{k+1}^{\star},
\]
where in the case of equality we choose to park. Define
\[
k^{\star}
=
\min\left\{
k\in\{0,1,\ldots,N-1\}:
c(k)\leq V_{k+1}^{\star}
\right\}.
\]
For every $k<k^{\star}$, by the definition of $k^{\star}$,
\[
c(k)>V_{k+1}^{\star},
\]
so the driver should continue even if space $k$ is free.

For every $k\geq k^{\star}$, since $c(k)$ is non-increasing and
$V_k^{\star}$ is non-decreasing,
\[
c(k)
\leq c(k^{\star})
\leq V_{k^{\star}+1}^{\star}
\leq V_{k+1}^{\star}.
\]
Therefore, from space $k^{\star}$ onward, the driver should park at the
first free space. Hence the optimal policy is a threshold policy.

If the set defining $k^{\star}$ is empty, the driver continues through
all parking spaces and parks in the garage; this is the degenerate threshold
case $k^{\star}=N$.
\end{proof}
